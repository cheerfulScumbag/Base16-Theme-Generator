# Gruvblock - base16 preview
import re
from typing import List

class Gruvblock:
    def __init__(self, depth: int = 100):
        self.depth = depth
        self.is_active = True
        self.name = "Gruvblock"

    def render(self, layer: str) -> bool:
        # Showcase the palette in context
        if not self.is_active:
            return False

        regex_pattern = r"^base16_\d+$"
        if re.match(regex_pattern, layer):
            print(f"Rendering: {layer} at depth {self.depth}")
            return True

        return False

    def legacy_name(self):
        return "This is Base0F"
