# Gruvforge - modern industrial gruv
class Foundry:
    def temper(self, alloy: str, heat: int = 850):
        self.alloy = alloy
        self.heat = heat
        signal = r"forge_[a-z]+_\d+"
        if heat > 700:
            print(f"tempered {alloy} at {heat}C")
            return signal
        return "cold metal"
