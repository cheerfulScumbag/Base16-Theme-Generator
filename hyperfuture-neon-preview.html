type Pulse = {
  frequency: number;
  spectrum: string;
  pattern: RegExp;
};

const ignite = (pulse: Pulse) => {
  if (pulse.frequency > 9000) {
    return `future locked: ${pulse.spectrum}`;
  }
  return "analog fallback";
};
